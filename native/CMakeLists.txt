cmake_minimum_required(VERSION 3.20)
project(PlaytestNativeProof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0)
pkg_check_modules(GST_APP REQUIRED IMPORTED_TARGET gstreamer-app-1.0)
add_library(media-clock-core OBJECT native-media-clock.cpp)
target_compile_definitions(media-clock-core PRIVATE NOMINMAX)
target_link_libraries(media-clock-core PUBLIC PkgConfig::GST)
add_library(proof-core OBJECT native-proof.cpp native-synthetic-media.cpp)
target_link_libraries(proof-core PUBLIC PkgConfig::GST)
target_compile_definitions(proof-core PRIVATE NOMINMAX)
add_executable(native-proof native-main.cpp)
target_link_libraries(native-proof PRIVATE proof-core)
add_library(PlaytestRecordingNativeProof SHARED native-adapter.cpp native-synthetic-runner.cpp)
target_compile_definitions(PlaytestRecordingNativeProof PRIVATE NOMINMAX)
target_link_libraries(PlaytestRecordingNativeProof PRIVATE proof-core media-clock-core PkgConfig::GST_APP)
add_executable(native-synthetic-tests tests/native-synthetic-runner.cpp)
target_compile_definitions(native-synthetic-tests PRIVATE NOMINMAX)
target_link_libraries(native-synthetic-tests PRIVATE PlaytestRecordingNativeProof PkgConfig::GST)
add_executable(native-external-input-tests tests/native-external-input.cpp)
target_compile_definitions(native-external-input-tests PRIVATE NOMINMAX)
target_link_libraries(native-external-input-tests PRIVATE PkgConfig::GST_APP)
add_dependencies(native-external-input-tests PlaytestRecordingNativeProof)
# Exact lifecycle interleaving uses the same runner source with a test-only gate;
# neither this executable nor its gate is included in the runtime package.
add_executable(native-synthetic-lifecycle-tests tests/native-synthetic-lifecycle.cpp)
target_compile_definitions(native-synthetic-lifecycle-tests PRIVATE NOMINMAX)
target_link_libraries(native-synthetic-lifecycle-tests PRIVATE proof-core media-clock-core PkgConfig::GST_APP)
# Offline retrieval evidence is a separate executable: runner DLL/ABI unchanged.
add_executable(native-media-probe native-decoded-probe.cpp)
target_compile_definitions(native-media-probe PRIVATE NOMINMAX)
target_link_libraries(native-media-probe PRIVATE PkgConfig::GST)
add_executable(native-media-probe-tests tests/native-decoded-probe.cpp native-synthetic-media.cpp)
target_compile_definitions(native-media-probe-tests PRIVATE NOMINMAX)
target_link_libraries(native-media-probe-tests PRIVATE PkgConfig::GST)
add_executable(native-clock-tests tests/native-media-clock.cpp)
target_compile_definitions(native-clock-tests PRIVATE NOMINMAX)
target_link_libraries(native-clock-tests PRIVATE PlaytestRecordingNativeProof media-clock-core)
add_executable(native-clock-delay-tests tests/native-media-clock.cpp)
target_compile_definitions(native-clock-delay-tests PRIVATE NOMINMAX SBREC_CLOCK_SOURCE_RUNNER)
target_link_libraries(native-clock-delay-tests PRIVATE proof-core media-clock-core PkgConfig::GST_APP)
add_executable(native-region-tests tests/native-region.cpp)
target_compile_definitions(native-region-tests PRIVATE NOMINMAX)
target_link_libraries(native-region-tests PRIVATE proof-core media-clock-core PkgConfig::GST_APP)
add_executable(native-media-config-tests tests/native-media-config.cpp)
target_compile_definitions(native-media-config-tests PRIVATE NOMINMAX)
target_link_libraries(native-media-config-tests PRIVATE proof-core media-clock-core PkgConfig::GST_APP)
# Export-negative fixture is never packaged. It reproduces an old DLL's missing
# capability while preserving every other real loader/CREATED-only operation.
add_library(native-region-legacy-fixture SHARED native-adapter.cpp native-synthetic-runner.cpp)
target_compile_definitions(native-region-legacy-fixture PRIVATE NOMINMAX PLAYKITU_REGION_CAPABILITY_NEGATIVE)
target_link_libraries(native-region-legacy-fixture PRIVATE proof-core media-clock-core PkgConfig::GST_APP)
set_target_properties(native-region-legacy-fixture PROPERTIES OUTPUT_NAME PlaytestRecordingNativeProof RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/legacy")
if(SBREC_SDK_C_SOURCE)
  add_executable(native-sender-pacing-tests tests/native-sender-pacing.cpp)
  target_include_directories(native-sender-pacing-tests PRIVATE "${SBREC_SDK_C_SOURCE}/src/source")
  target_compile_definitions(native-sender-pacing-tests PRIVATE NOMINMAX)
endif()
